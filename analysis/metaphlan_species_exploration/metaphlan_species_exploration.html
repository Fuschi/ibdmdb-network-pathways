library(tidyverse)
library(mgnet)
library(compositions)
ibdm <- mgnet(
  
  rela = read_tsv("../../results/metaphlan/metaphlan_species_abundance_matrix.tsv",
                 show_col_types = FALSE) %>%
    column_to_rownames("sample_id") %>%
    as.matrix(),
  
  meta = read_tsv("../../data/metadata_ena_sra/processed/ibdmdb_metadata_runs_mgx.tsv",
                   show_col_types = FALSE) %>%
    select(run_accession, host_subject_id, host_disease,
           host_sex, host_age, collection_date) %>%
    left_join(read_tsv("../../results/kneaddata_summary.tsv",
                       show_col_types = FALSE),
              by = "run_accession") %>%
    mutate(host_subject_id = as.factor(host_subject_id)) %>%
    arrange(run_accession) %>%
    column_to_rownames("run_accession"),
  
  taxa = read_tsv("../../results/metaphlan/metaphlan_species_taxonomy_matrix.tsv",
                   show_col_types = FALSE) %>%
    column_to_rownames("species"))
meta(ibdm, "tbl") %>%
  mutate(sample_id = as.character(sample_id),
         host_subject_id = as.character(host_subject_id),
         sample_id = fct_reorder(sample_id, raw_pairs, .desc = TRUE)) %>%
  rename(cleaned = clean_pairs,
         trimmed_removed = pairs_removed_by_trim,
         human_removed = pairs_removed_after_trim) %>%
  select(sample_id, host_subject_id, host_disease, host_sex,
    raw_pairs, cleaned, trimmed_removed, human_removed) %>%
  pivot_longer(cols = c(cleaned, trimmed_removed, human_removed),
               names_to = "read_class", values_to = "read_pairs") %>%
  mutate(read_class = factor(read_class,
                             levels = c("human_removed", "trimmed_removed", "cleaned"),
                             labels = c("Human removed", "Removed by trimming", "Cleaned"))) %>%
  ggplot(aes(x = sample_id, y = read_pairs, fill = read_class)) +
  geom_col(width = .95) +
  scale_y_continuous(labels = scales::label_number(scale_cut = scales::cut_short_scale())) +
  scale_fill_brewer(palette = "Set1") +
  theme_bw() +
  labs(x = "Sample ID", y = "Raw read pairs", fill = "Read class") +
  theme(axis.text.x = element_blank(),
        axis.ticks.x = element_blank(),
        panel.grid.major.x = element_blank(),
        legend.position = "top")

pca_clr <- prcomp(mgnet::norm(ibdm), center = TRUE, scale. = FALSE)

var_explained <- round(100 * pca_clr$sdev^2 / sum(pca_clr$sdev^2), 1)

pca_clr$x[, 1:2] %>%
  as_tibble(rownames = "sample_id") %>%
  left_join(meta(ibdm, "tbl"), by = "sample_id") %>%
